using System;

#if !API_DRIVER_SYSTEM_REFLECTION_METADATA && !API_DRIVER_MONO_CECIL
public delegate T TheDelegate<T>(T t);
#endif

public class GenericClass<T>
{
    public T theField;
    public T TheProperty  { get; set; }
    public T TheMethod(T p) { return p; }

#if !API_DRIVER_SYSTEM_REFLECTION_METADATA && !API_DRIVER_MONO_CECIL
    public delegate T TheDelegate(T t);
#endif
}

public struct GenericStruct<T> 
{
    public T theField;
    public T TheProperty  { get; set; }
    public T TheMethod(T p) { return p; }
    
#if !API_DRIVER_SYSTEM_REFLECTION_METADATA && !API_DRIVER_MONO_CECIL
    //TODO: Generic delegates are not supported yet
    public delegate T TheDelegate(T t);
#endif
}

public interface IGenericInterface<T> 
{
    T TheProperty  { get; set; }
    T TheMethod(T p);

    event EventHandler<T> TheEventT;
}
